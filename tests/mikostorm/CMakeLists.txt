cmake_minimum_required(VERSION 3.21)
project(MikoStormRegressionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
find_package(Threads REQUIRED)
add_executable(core_regression core_regression.cpp)
target_include_directories(core_regression PRIVATE ../../indra/newview)
target_link_libraries(core_regression PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(core_regression PRIVATE /W4 /WX /Z7)
endif()
add_test(NAME core_regression COMMAND core_regression)
# Compile the production logger with small viewer-service doubles. Only its PCH include changes.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../indra/newview/fsmikoavatarlog.cpp" logger_source)
string(REPLACE "#include \"llviewerprecompiledheaders.h\"" "#include \"viewer_test_stubs.h\"" logger_source "${logger_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/avatarlog_under_test.cpp" "${logger_source}")
add_executable(avatarlog_regression avatarlog_regression.cpp "${CMAKE_CURRENT_BINARY_DIR}/avatarlog_under_test.cpp")
find_path(MIKO_BOOST_INCLUDE_DIR boost/signals2.hpp HINTS
  "${CMAKE_CURRENT_SOURCE_DIR}/../../build-vc170-64-avx2/packages/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../build-vc170-64-legacy/packages/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../build-vc170-64/packages/include"
  REQUIRED)
target_include_directories(avatarlog_regression PRIVATE stubs ../../indra/newview "${MIKO_BOOST_INCLUDE_DIR}")
target_link_libraries(avatarlog_regression PRIVATE Threads::Threads)
add_test(NAME avatarlog_regression COMMAND avatarlog_regression)
set_tests_properties(avatarlog_regression PROPERTIES TIMEOUT 30)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../indra/newview/llcamerajointselector.h" camera_header)
string(REPLACE "#include \"llagentcamera.h\"" "#include \"camera_services_stub.h\"" camera_header "${camera_header}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/llcamerajointselector.h" "${camera_header}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../indra/newview/llcamerajointselector.cpp" camera_source)
string(REPLACE "#include \"llviewerprecompiledheaders.h\"" "#include \"camera_services_stub.h\"" camera_source "${camera_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/camera_under_test.cpp" "${camera_source}")
add_executable(camera_regression camera_regression.cpp "${CMAKE_CURRENT_BINARY_DIR}/camera_under_test.cpp")
target_include_directories(camera_regression PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" camera_stubs)
add_test(NAME camera_regression COMMAND camera_regression)
